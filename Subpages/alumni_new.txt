  <script>
    function toggleMenu() {
      const menu = document.getElementById('menu');
      menu.classList.toggle('open');
    }

    // === Alumni-only logic ===
    document.addEventListener('DOMContentLoaded', () => {
      const courseEl = document.getElementById('courseFilter');
      const yearEl = document.getElementById('yearFilter');
      const qEl = document.getElementById('alumniSearch');
      const container = document.getElementById('alumniContainer');

      // ---- DOM sanity check / self-heal ----
      // Any direct child missing 'alumni-card' but looks like a card? Add the class.
      [...container.children].forEach((node, i) => {
        if (node.nodeType === 1) {
          const hasClass = node.classList.contains('alumni-card');
          const hasData = node.hasAttribute('data-course') && node.hasAttribute('data-year');
          if (!hasClass && hasData) {
            console.warn(`[alumni] Child #${i+1} had data-course/year but no 'alumni-card' class. Adding it automatically.`);
            node.classList.add('alumni-card');
          }
        }
      });

      const cards = Array.from(container.querySelectorAll('.alumni-card'));
      const exportControls = document.getElementById('exportControls');
      const selectAllEl = document.getElementById('selectAll');
      const resultNote = document.getElementById('resultNote');
      const emptyState = document.getElementById('emptyState');

      updateExportVisibility(0);

      document.getElementById('searchBtn').addEventListener('click', runSearch);
      document.getElementById('resetBtn').addEventListener('click', () => {
        courseEl.value = 'all';
        yearEl.value = 'all';
        qEl.value = '';
        cards.forEach(c => c.classList.remove('show'));
        selectAllEl.checked = false;
        updateExportVisibility(0);
        resultNote.textContent = '';
        emptyState.style.display = '';
      });

      function runSearch() {
        const course = courseEl.value.trim().toLowerCase();
        const year = yearEl.value.trim().toLowerCase();
        const q = qEl.value.trim().toLowerCase();

        const hasAnyInput = (course !== 'all') || (year !== 'all') || (q.length > 0);

        if (!hasAnyInput) {
          cards.forEach(c => c.classList.remove('show'));
          updateExportVisibility(0);
          resultNote.textContent = 'Please provide a search term and/or select a filter.';
          emptyState.style.display = '';
          return;
        }

        let visibleCount = 0;
        cards.forEach(card => {
          const cCourse = (card.getAttribute('data-course') || '').toLowerCase();
          const cYear   = (card.getAttribute('data-year') || '').toLowerCase();
          const text    = card.innerText.toLowerCase();

          const courseOk = (course === 'all') || (cCourse === course);
          const yearOk   = (year === 'all')   || (cYear === year);
          const qOk      = !q || text.includes(q);

          const show = courseOk && yearOk && qOk;
          card.classList.toggle('show', show);

          if (!show) {
            const cb = card.querySelector('.alumni-checkbox');
            if (cb) cb.checked = false;
          }
          if (show) visibleCount++;
        });

        selectAllEl.checked = false;
        updateExportVisibility(visibleCount);
        resultNote.textContent = visibleCount > 0
          ? `Showing ${visibleCount} result${visibleCount>1?'s':''}.`
          : 'No matching alumni found.';
        emptyState.style.display = 'none';
      }

      function updateExportVisibility(visibleCount) {
        exportControls.setAttribute('aria-hidden', visibleCount > 0 ? 'false' : 'true');
      }
    });

    // Select-all only affects VISIBLE cards
    function toggleSelectAll() {
      const checked = document.getElementById('selectAll').checked;
      document.querySelectorAll('.alumni-card').forEach(card => {
        if (getComputedStyle(card).display !== 'none') {
          const cb = card.querySelector('.alumni-checkbox');
          if (cb) cb.checked = checked;
        }
      });
    }

    // Export only VISIBLE + SELECTED cards
    async function exportSelectedAlumni(mode) {
      const visibleSelectedCards = [...document.querySelectorAll('.alumni-card')]
        .filter(card => getComputedStyle(card).display !== 'none')
        .filter(card => card.querySelector('.alumni-checkbox')?.checked);

      if (visibleSelectedCards.length === 0) {
        alert("Please select at least one alumnus from the visible results to export.");
        return;
      }

      const rows = visibleSelectedCards.map(card => {
        const name = card.querySelector("h3")?.innerText || '';
        const course = (card.getAttribute("data-course") || '').toUpperCase();
        const year = card.getAttribute("data-year") || '';
        const ps = card.querySelectorAll("p");
        const position = ps[2]?.innerText.replace("Current Position: ", "") || '';
        const contact = ps[3]?.innerText.replace("Contact: ", "") || '';
        return [name, course, year, position, contact];
      });

      if (mode === 'screen') {
        let tableHTML = `<table border="1" cellspacing="0" cellpadding="6">
          <thead><tr>
            <th>Name</th><th>Course</th><th>Year</th><th>Position</th><th>Contact</th>
          </tr></thead><tbody>`;
        rows.forEach(r => {
          tableHTML += `<tr>
            <td>${r[0]}</td><td>${r[1]}</td><td>${r[2]}</td><td>${r[3]}</td><td>${r[4]}</td>
          </tr>`;
        });
        tableHTML += "</tbody></table>";

        const newWin = window.open("", "ExportedAlumni");
        newWin.document.write(`<html><head><title>Exported Alumni</title>
          <style>
            body { font-family: ui-sans-serif, system-ui, -apple-system, Segoe UI, Roboto, Arial, sans-serif; padding: 20px; }
            table { border-collapse: collapse; width: 100%; }
            th, td { border: 1px solid #aaa; padding: 8px; text-align: left; }
            th { background: #d0eaff; }
          </style></head><body>
          <h2>Exported Alumni</h2>${tableHTML}</body></html>`);
        newWin.document.close();
        newWin.focus();
      }

      if (mode === 'pdf') {
        const { jsPDF } = window.jspdf;
        const doc = new jsPDF();

        doc.setFontSize(14);
        doc.text("Exported Alumni", 14, 15);

        doc.autoTable({
          head: [["Name", "Course", "Year", "Position", "Contact"]],
          body: rows,
          startY: 25,
          styles: { fontSize: 10, cellPadding: 3, overflow: 'linebreak' },
          headStyles: { fillColor: [102,207,255], textColor: [0,51,68], fontStyle: 'bold' },
          alternateRowStyles: { fillColor: [245,250,255] },
          margin: { left: 14, right: 14 },
        });

        doc.save("Exported_Alumni.pdf");
      }
    }
  </script>
